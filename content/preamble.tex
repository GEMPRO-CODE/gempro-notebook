\newcommand{\addtemplate}[1]{
	\ifdefined\preparedtemplates%
	% Preserve the whitespace formerly produced around the shell commands.
	\space%
	\input{./out/prepared/#1.tex}
	\space%
	\else%
	\immediate\write18{python3 ./scripts/prepare.py ./out/\jobname.prepared-template.tex #1 < ./content/#1}
	\input{./out/\jobname.prepared-template.tex}
	\immediate\write18{rm -f ./out/\jobname.prepared-template.tex}
	\fi%
}

% A description shares its heading's line; an empty description keeps a block heading.
% Group the run-in style so category headings and the appendix stay standalone.
\long\def\notebookentry#1#2#3{%
    \begingroup
    \if\relax\detokenize{#2}\relax
        \subsection{#1}
    \else
        \titleformat{\subsection}[runin]{\normalsize\bfseries}{}{0pt}{}[.]
        \titlespacing*{\subsection}{0pt}{0.65ex plus 0.15ex minus 0.1ex}{0.5em}
        \subsection{#1}#2\par
    \fi
    #3\par
    \endgroup
}

\long\def\templateentry#1#2#3#4{%
    \ifextended
        \notebookentry{#1}{#3}{\addtemplate{#4}}
    \else
        \notebookentry{#1}{#2}{\addtemplate{#4}}
    \fi
}

\long\def\mintedentry#1#2#3#4#5{%
    \ifextended
        \notebookentry{#1}{#3}{\inputminted{#4}{#5}}
    \else
        \notebookentry{#1}{#2}{\inputminted{#4}{#5}}
    \fi
}
